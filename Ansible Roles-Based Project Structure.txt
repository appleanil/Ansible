1. ansible.cfg 
--------------
[defaults]
inventory = inventory/hosts.ini
roles_path = roles
host_key_checking = False
remote_user = ec2-user


2. inventory
------------
[webservers]
web1
web2

[dbservers]
db1


3. inventory/dev.ini
--------------------
[webservers]
dev-web1


4. inventory/prod.ini
---------------------
[webservers]
prod-web1
prod-web2


group_vars
==========
1. group_vars/all.yml
---------------------

----------------------------

3. group_vars/dbservers.yml
---------------------------

host vars
=========
1. host_vars/web1.yml
----------------------

2. host_vars/db1.yml
--------------------
mysql_root_password: root123


playbooks
=========
playbooks/site.yml
------------------
- hosts: all
  become: true
  roles:
    - common

playbooks/web.yml
-----------------
- hosts: webservers
  become: true
  roles:
    - httpd

playbooks/db.yml
----------------
- hosts: dbservers
  become: true
  roles:
    - mysql

roles/common
============
roles/common/defaults/main.yml
------------------------------
common_packages:
  - vim
  - git

roles/common/tasks/main.yml
---------------------------
- name: Install common packages
  yum:
    name: "{{ common_packages }}"
    state: present

roles/common/handlers/main.yml
------------------------------
- name: reload ssh
  service:
    name: sshd
    state: restarted

roles/common/README.md
----------------------
Role: common  
Purpose: Install base packages and configs


roles/mysql
===========
roles/mysql/defaults/main.yml
-----------------------------
mysql_pkg: mysql-server


roles/mysql/tasks/main.yml
--------------------------
- name: Install MySQL
  yum:
    name: "{{ mysql_pkg }}"
    state: present


🐳 roles/docker
===============
roles/docker/tasks/main.yml
---------------------------
- name: Install Docker
  yum:
    name: docker
    state: present

- name: Start Docker
  service:
    name: docker
    state: started
